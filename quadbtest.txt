package main

import "fmt"

func QuadB(x, y int) {
	if x <= 0 || y <= 0 {
		return
	}

	for i := 0; i < y; i++ { // loop grammis (i = y) // j = x)
		for j := 0; j < x; j++ { // loop stilis

			if i == 0 && j == 0 { // an vrisketai stin pano aristeri gonia (1os xaraktiras (1,1)) - && (me ta && orizoume tis 4 gonies)
				fmt.Print("/")
			} else if i == 0 && j == x-1 { // an vrisketai stin pano dexia gonia (last charachter) - &&
				fmt.Print("\\")
			} else if i == y-1 && j == 0 { // an vrisketai sthn kato (i == y-1) aristeri gonia (j == 0) - &&
				fmt.Print("\\")
			} else if i == y-1 && j == x-1 { // an vrisketai sthn kato (i == y-1) dexia gonia (j == x-1) - &&
				fmt.Print("/")
			} else if i == 0 || i == y-1 { // an vrisketai stin korifi Top (i == 0) i kato Bottom (i == y-1) - ||
				fmt.Print("*")
			} else if j == 0 || j == x-1 { // an vrisketai terma aristera (j == 0) i terma dexia (j == x-1) - ||
				fmt.Print("*")
			} else { // an den simvainei tipota apo ta parapano tote
				fmt.Print(" ")
			}
		}
		fmt.Println()
	}
}

func main() {
	QuadB(5, 5)
}
